### 7.1 ★★ `active` 漏写（**最强候选**，指向"渲染不出来"）

### 11.6.1 复现本节的仪器（**被并发代理删了，代码在此**，拷回 `tools/` 即可用）
```lua
-- tools/_audit_globals.lua（动态幻影变量探测）核心：
local WHITELIST = { game=true, script=true, print=true, printerr=true, Enum=true,
  Color=true, ColorValue=true, FORCE_LEVEL=true, math=true, table=true, string=true,
  os=true, io=true, debug=true, coroutine=true, utf8=true, tostring=true, tonumber=true,
  type=true, pairs=true, ipairs=true, pcall=true, xpcall=true, error=true, assert=true,
  select=true, next=true, setmetatable=true, getmetatable=true, rawget=true, rawset=true }
local hits, order, in_hook = {}, {}, false
setmetatable(_G, { __index = function(_, k)
  if in_hook then return nil end
  in_hook = true
  local key = tostring(k)
  if not WHITELIST[key] then
    local info = debug.getinfo(2, "Sl")
    local id = ((info and info.short_src) or "?") .. ":" .. tostring((info and info.currentline) or -1) .. "  " .. key
    if hits[id] == nil then hits[id] = 0; order[#order + 1] = id end
    hits[id] = hits[id] + 1
  end
  in_hook = false
  return nil
end })
-- 然后 loadfile 各测试并 pcall 执行；跑完打印 order（⚠️ 测试会 hijack print ⇒ 先 `local out = print`；

```

收集 `local`/形参/for 变量 → 查 `IDENT:` 与 `IDENT.` 的接收者能否解析。
⚠️ 两个坑都写进注释了：`local function f(...)` 必须在 `function` 上**停住**（`i = j` 而不是 `i = j + 1`），
否则作用域栈从此错位、满屏假阳性（第一版 355 条 → 修好后 2 条）。
